\definecolor{imageblue}{RGB}{36,91,125}
\definecolor{fourierorange}{RGB}{181,103,42}
\definecolor{referencegray}{RGB}{66,66,66}
\definecolor{axisgray}{RGB}{70,70,70}

\begin{document}
\begin{minipage}{7.0in}
\centering
\captionsetup{font=footnotesize,labelfont=bf,skip=3pt,width=7.0in}
\pgfplotstableread[col sep=comma]{figure_auc_ap_landscape_data.csv}\landscapedata

\begin{tikzpicture}
\begin{axis}[
    width=6.25in,
    height=3.20in,
    scale only axis,
    xlabel={Test AUC},
    ylabel={Test average precision (AP)},
    xmin=0.33,
    xmax=0.83,
    ymin=0.03,
    ymax=0.23,
    xtick={0.35,0.40,0.45,0.50,0.55,0.60,0.65,0.70,0.75,0.80},
    ytick={0.04,0.08,0.12,0.16,0.20},
    tick label style={font=\scriptsize,text=axisgray},
    label style={font=\footnotesize,text=axisgray},
    axis line style={draw=axisgray,line width=0.45pt},
    tick style={draw=axisgray,line width=0.35pt},
    grid=major,
    major grid style={draw=black!10,line width=0.25pt},
    legend style={
      font=\scriptsize,
      draw=none,
      fill=white,
      fill opacity=0.94,
      text opacity=1,
      at={(0.5,1.035)},
      anchor=south,
      inner sep=1.5pt,
      column sep=8pt,
    },
    legend cell align=left,
    clip=true,
]

% Fixed real magnitude baseline, drawn beneath the observations.
\addplot[referencegray,dashed,line width=0.75pt,forget plot]
  coordinates {(0.7268014932989552,0.03) (0.7268014932989552,0.23)};
\addplot[referencegray,dashed,line width=0.75pt,forget plot]
  coordinates {(0.33,0.10628203909007349) (0.83,0.10628203909007349)};

% Marker shape and hue both encode domain; coordinates are not jittered.
\addplot+[
    only marks,
    mark=*,
    mark size=1.35pt,
    draw=imageblue,
    fill=imageblue,
    opacity=0.52,
    line width=0.25pt,
    restrict expr to domain={\thisrow{domain_code}}{0:0},
    forget plot,
] table[x=test_auc,y=test_ap] {\landscapedata};

\addplot+[
    only marks,
    mark=triangle,
    mark options={rotate=0},
    mark size=1.55pt,
    draw=fourierorange,
    fill=white,
    opacity=0.70,
    line width=0.45pt,
    restrict expr to domain={\thisrow{domain_code}}{1:1},
    forget plot,
] table[x=test_auc,y=test_ap] {\landscapedata};

% Explicit legend samples prevent the two reference lines from inheriting
% marker styles and keep the legend interpretable in grayscale.
\addlegendimage{legend image code/.code={\draw[referencegray,dashed,line width=0.75pt] (0cm,0cm) -- (0.55cm,0cm);}}
\addlegendentry{Real magnitude baseline}
\addlegendimage{legend image code/.code={\draw[imageblue,fill=imageblue] (0.275cm,0cm) circle (1.35pt);}}
\addlegendentry{Image domain}
\addlegendimage{legend image code/.code={\draw[fourierorange,fill=white,line width=0.45pt] (0.275cm,-0.02cm) -- (0.275cm,0.08cm) -- (0.365cm,-0.08cm) -- cycle;}}
\addlegendentry{Fourier domain}
\end{axis}
\end{tikzpicture}
